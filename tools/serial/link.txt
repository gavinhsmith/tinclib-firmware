# Link smoke test: framing, handshake, reply cache and error replies.
# Needs no Wi-Fi.
#   serial-passthrough -a COM5 -p tools/serial/tinc.spec < tools/serial/link.txt
#
# Payloads are little-endian hex, built from the layouts in protocol.h.
# Every new command gets a new seq; reusing one asks for the cached reply.

wait 2000                   # opening the port usually resets the board; let it boot

# After a reset, anything before HELLO gets ERR_NO_HELLO: flags=5 payload=[02]
pkt type=2 seq=1
expect 500 type=2 seq=1

# HELLO 0.1, our max_payload 256 -> ESP 0.1, caps 0, max_payload 1024 (00 04)
pkt type=1 seq=2 payload=000100000001
expect 500 flags=1 type=1 seq=2 len=10 payload=[00 01 00 00 00 04

# HELLO with the wrong minor -> ERR_VERSION (03) + the ESP's own version
pkt type=1 seq=3 payload=000200000001
expect 500 flags=5 type=1 seq=3 len=3 payload=[03 00 01]

# STATUS: wifi_state, slot, rssi, ip[4], free_heap, req_state
pkt type=2 seq=4
expect 500 flags=1 type=2 seq=4 len=12
# same seq again: the cached reply comes back, the command isn't re-run
pkt type=2 seq=4
expect 500 flags=1 type=2 seq=4 len=12

# unknown type -> ERR_UNSUPPORTED (01)
pkt type=0x13 seq=5
expect 500 flags=5 type=19 seq=5 len=1 payload=[01]

# BODY_READ with no request -> ERR_BAD_STATE (06)
pkt type=0x21 seq=6 payload=00000000400000
expect 500 flags=5 type=33 seq=6 len=1 payload=[06]

# REQ_BEGIN https:// isn't in 0.1 -> ERR_UNSUPPORTED_SCHEME (09)
#   method 01, flags 00, timeout 00, content_len 00000000, url_len 1400, hdr_len 0000, url
pkt type=0x10 seq=7 payload=0100000000000014000000"https://example.com/"
expect 500 flags=5 type=16 seq=7 len=1 payload=[09]

# WIFI_LIST: 3 x (ssid_len, ssid), never passwords
pkt type=0x40 seq=8
expect 500 flags=1 type=64 seq=8

quit
