# One-time Wi-Fi setup. Copy to wifi.txt (gitignored), put in your network,
# run it once; the slot is saved in flash.
#   serial-passthrough -a COM5 -p tools/serial/tinc.spec < tools/serial/wifi.txt
#
# WIFI_SET payload: slot, ssid_len, "ssid", pass_len, "password"
# Lengths are hex: "HomeNet" is 07, "hunter22" is 08. Open network: pass_len 00.

wait 2000
pkt type=1 seq=1 payload=000100000001
expect 500 flags=1 type=1 seq=1

pkt type=0x41 seq=2 payload=0007"HomeNet"08"hunter22"
expect 2000 flags=1 type=65 seq=2

# the ESP rescans and joins; STATUS wifi_state 2 = connected (1 = still trying)
wait 10000
pkt type=2 seq=3
expect 500 flags=1 type=2 seq=3 len=12 payload=[02

quit
